# Replace `src/app/articles/[slug]/page.tsx` With This

```tsx
import fs from "fs";
import path from "path";
import matter from "gray-matter";
import MainNavbar from "@/components/MainNavbar";
import Image from "next/image";
import { notFound } from "next/navigation";

interface Props {
  params: Promise<{ slug: string }>;
}

function findMdxFile(slug: string): string | null {
  const articlesDir = path.join(process.cwd(), "content/articles");

  function searchDir(dir: string): string | null {
    const entries = fs.readdirSync(dir, { withFileTypes: true });

    for (const entry of entries) {
      const fullPath = path.join(dir, entry.name);

      if (entry.isDirectory()) {
        const found = searchDir(fullPath);
        if (found) return found;
      }

      if (entry.isFile() && entry.name === `${slug}.mdx`) {
        return fullPath;
      }
    }

    return null;
  }

  return searchDir(articlesDir);
}

export default async function ArticlePage({ params }: Props) {
  const { slug } = await params;

  const filePath = findMdxFile(slug);

  if (!filePath) {
    notFound();
  }

  const raw = fs.readFileSync(filePath, "utf8");

  const parsed = matter(raw);

  const frontmatter = parsed.data;
  const content = parsed.content;

  const paragraphs = content
    .split("\n\n")
    .filter((p) => p.trim().length > 0);

  return (
    <main
      style={{
        background: "#050816",
        color: "#f5f1e8",
        minHeight: "100vh",
      }}
    >
      <MainNavbar />

      {frontmatter.image && (
        <div
          style={{
            position: "relative",
            width: "100%",
            height: "520px",
            overflow: "hidden",
          }}
        >
          <Image
            src={frontmatter.image}
            alt={frontmatter.title || "Article image"}
            fill
            priority
            sizes="100vw"
            style={{
              objectFit: "cover",
              filter: "brightness(0.55)",
            }}
          />

          <div
            style={{
              position: "absolute",
              inset: 0,
              background:
                "linear-gradient(to top, rgba(5,8,22,0.98) 0%, rgba(5,8,22,0.55) 55%, rgba(5,8,22,0.15) 100%)",
            }}
          />

          <div
            style={{
              position: "absolute",
              left: "50%",
              bottom: "48px",
              transform: "translateX(-50%)",
              width: "100%",
              maxWidth: "1100px",
              padding: "0 32px",
            }}
          >
            <div
              style={{
                display: "flex",
                gap: "10px",
                marginBottom: "18px",
              }}
            >
              {frontmatter.featured && (
                <span className="badge badge-gold">Featured</span>
              )}

              {frontmatter.category && (
                <span className="badge badge-accent">
                  {frontmatter.category}
                </span>
              )}
            </div>

            <h1
              style={{
                fontFamily: "var(--font-family-playfair)",
                fontSize: "clamp(2.3rem, 5vw, 4.5rem)",
                lineHeight: 1.05,
                marginBottom: "20px",
                maxWidth: "900px",
                color: "white",
              }}
            >
              {frontmatter.title}
            </h1>

            <div
              style={{
                display: "flex",
                gap: "14px",
                color: "rgba(255,255,255,0.7)",
                textTransform: "uppercase",
                fontSize: "0.75rem",
                letterSpacing: "0.12em",
              }}
            >
              <span>{frontmatter.author}</span>
              <span>•</span>
              <span>{frontmatter.readTime}</span>
              <span>•</span>
              <span>{frontmatter.date}</span>
            </div>
          </div>
        </div>
      )}

      <article
        style={{
          maxWidth: "760px",
          margin: "0 auto",
          padding: "72px 32px 120px",
          fontFamily: "var(--font-family-serif)",
          fontSize: "1.18rem",
          lineHeight: 1.95,
          color: "#d6d1c7",
        }}
      >
        {paragraphs.map((para, i) => {
          if (para.startsWith("## ")) {
            return (
              <h2
                key={i}
                style={{
                  fontFamily: "var(--font-family-playfair)",
                  fontSize: "2rem",
                  marginTop: "2.8em",
                  marginBottom: "0.8em",
                  lineHeight: 1.2,
                  color: "#ffffff",
                }}
              >
                {para.replace("## ", "")}
              </h2>
            );
          }

          if (para.trim() === "---") {
            return (
              <hr
                key={i}
                style={{
                  border: "none",
                  borderTop: "1px solid rgba(255,255,255,0.08)",
                  margin: "3em 0",
                }}
              />
            );
          }

          return (
            <p key={i} style={{ marginBottom: "1.7em" }}>
              {para}
            </p>
          );
        })}
      </article>
    </main>
  );
}
```
